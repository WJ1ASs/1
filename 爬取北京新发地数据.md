import requests
import csv
import time
url = "http://wap.xinfadi.com.cn/getPriceData.html"
headers = {
    "User-Agent":( "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"
    " AppleWebKit/537.36 (KHTML, like Gecko)"
    " Chrome/155.0.0.0 Safari/537.36 Edg/155.0.0.0")
}
now_current = 1
with open("北京新发地菜价.csv","w",encoding="utf-8",newline="") as f:
    writer = csv.writer(f)
    writer.writerow(["一级分类", "二级分类", "品名", "最低价", "平均价", "最高价", "产地", "单位", "日期"])
    while 10086:
        data = {
            "limit": 20,
            "current": now_current
        }
        res = requests.post(url, headers=headers, data=data)
        kk = res.json()
        jj = kk["list"]
        if not jj:
            print("全部抓取成功")
            break

        for i in jj:
            row = [
                i["prodCat"],
                i["prodPcat"],
                i["prodName"],
                i["lowPrice"],
                i["avgPrice"],
                i["highPrice"],
                i["place"],
                i["unitInfo"],
                i["pubDate"]
            ]
            writer.writerow(row)
            print(row)
        print(f"---第 {now_current} 页抓取完成---")
        now_current = now_current + 1
        time.sleep(1.5)
